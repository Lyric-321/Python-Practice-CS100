-------ex1.py-------
""" operations with float point numbers (decimal/real number) """
x = input("Enter a real number: ")
x = float(x)
y = input("Enter a real number: ")
y = float(y)

s = x + y
print(f"{x} + {y} = {s}")

d = x - y
print(f"{x} - {y} = {d}")

p = x * y
print(f"{x} * {y} = {p}")

q = x / y
print(f"{x} / {y} = {q}")

pw = x ** y
print(f"{x} ** {y} = {pw}")

-------ex6.py-------
''' composite assignments '''
x = input("Enter a real value x = ")
x = float(x)
print(f"x = {x}")

k = input("Enter a real value k = ")
k = float(k)
print(f"k = {k}")

x += k     # increment x by k
print(f"x += k changes x to {x}")

x -= k     # decrease x by k
print(f"x -= k changes x to {x}")

x *= k     # multiply x by k
print(f"x *= k changes x to {x}")

x /= k     # divide x by k
print(f"x /= k changes x to {x}")
