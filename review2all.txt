def longdiv(a,b):
    q = a//b
    r = a % b
    return (q,r)

def isInt(s):
    try:
        int(s)
        return True
    except ValueError:
        return False


def main():
    while True:
        ans = input("Enter q to quit or a pair of 2 integers: ")
        if ans == 'q':
            break
        t = ans.split(' ')
        if len(t) != 2:
            print('Invalid input...')
            continue    #continue has to be a part of if statement, not by itself
        if isInt(t[0]) == False or isInt(t[1]) == False:
            print('Invalid input...')
            continue
        a = int(t[0])
        b = int(t[1])
        if b == 0:
            print("Division by 0 is not defined.")
            continue
        t = longdiv(a,b)
        print(f"{a} = {b}*{t[0]}+{t[1]}")

if __name__ == '__main__':
    main()

def range(v):
    mx = mn = v[0]
    for i in v:
        if mx < i:

        if mn > i:

def median(v):
   v.sort()
   if len(v) % 2 == 1:
       return v[(len(v)-1)//2]

import random as rd
def rps():
    dict = {1:'Scissors', 2:'Paper', 3:'Rock'}
    you = int(input(" 1 for 'Scissors', 2 for 'Paper', or 3 for'Rock': "))
    mac = rd.randint(1,3)
    print(f"You: {dict[you]} vs Machine: {dict[mac]}")

    winning = [ (2,1), (3,2), (1,3) ]  #(you,mac)
    drawing = [ (1,1), (2,2), (3,3) ]
    losing  = [ (1,2), (2,3), (3,1) ]

    if (you,mac) in winning:
        print(f'You win: {dict[you]} beats {dict[mac]}.')
        return 'win'
    elif (you,mac) in drawing:
        print(f'The game is a draw: {dict[you]} vs {dict[mac]}.')
        return 'draw'
    else:
        print(f'You lose: {dict[mac]} beats {dict[you]}.')
        return 'lose'

def main():
    print('Rock-Paper-Scissors\n')
    while True:
        s = input('Enter q to quit or anything else to play: ')
        if s == 'q':
            break
        wincount = losecount = 0
        while wincount < 2 and losecount < 2:
            result = rps()
            if result == 'win':
                wincount += 1
            if result == 'lose':
                losecount += 1
            print(f'Score You vs Mac: {wincount}-{losecount}')
        if wincount == 2:
            print('You won the game!')
        else:
            print('You lost the game!')
        print('-----------------------------------------\n')

if __name__ == '__main__':
    main()

def play(secret):
    i = 1
    while ( i <= 5 ):
        ans = input(f'Guess {i} is: ')
        ans = int(ans)
        if ans == secret:
            return True
        i += 1
    return False

def main():
    print('The Game is: Guess a number from 1 to 20 in 5 tries...' )
    while True:
        s = input('Enter q to quit or anything else to continue: ' )
        if s == 'q':
            break
        secret = rd.randint(1,20)
        win = play(secret)
        if win:
            print("Congratulation: You won the game!")
        else:
            print(f'You lost. The secret number is {secret}.')

    main()
