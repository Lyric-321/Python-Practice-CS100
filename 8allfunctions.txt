def add(list):
    s = 0
    for item in list:
        s += item
    return s

def value(v):
    try:
        float(v)
        return True
    except ValueError:
        return False

def main():
    while True:
        s = input("Enter q to quit or a list of values:\n ")
        if s == "q":
            break
        list = []
        for v in s.split(' '):
            if value(v):
                list.append(float(v))
        print(list)
        s = add(list)
        print(f"The sum of the values is {s}. ")

main()

# compute s = 1 + 2 + 3 + .. + n
def sum(n):
    s = 0
    for i in range(1,n+1):
        s += i
    return s

def main():
    while True:
        ans = input("Enter q to quit or a positive integer n: ")
        if ans == "q":
            break
        n = int(ans)
        s = sum(n)
        # print the sum
        if n > 10:
            print(f"1 + 2 + . . . + {n} = {s}")
        else:
            print("1", end="")
            for i in range(2,n+1):
                print(f" + {i}", end="")
            print(f" = {s}")

main()

# compute s = 1/1^2 + 1/2^2 + 1/3^2 + .. + 1/n^2

def sum(n):
    s = 0
    for i in range(1,n+1):
        s += 1.0/i/i
    return s

def main():
    while True:
        ans = input("Enter q to quit or a positive integer n: ")
        if ans == "q":
            break
        n = int(ans)
        s = sum(n)
        # print the sum
        if n > 10:
            print(f"1 + 2/2^2 + . . . + 1/{n}^2 = {s}")
        else:
            print("1", end="")
            for i in range(2,n+1):
                print(f" + 1/{i}^2", end="")
            print(f" = {s}")
            
if __name__ == "__main__":
    main()

    if 90 <= g:
        letter = "A"
    elif 80 <= g < 90:
        letter = "B"
    elif 70 <= g < 80:
        letter = "C"
    elif 60 <= g < 70:
        letter = "D"
    else:
        letter = "F"
    return letter

def main():
    print("Enter q to quit at anytime...")
    while True:
        g = input("Enter a grade between 0 and 100: ")
        if g == 'q':
            break
        g = float(g)
        lg = lettergrade(g)
        print(f"grade of {g} = letter grade of {lg}")

main()

    try:
        float(s)

    except ValueError:
        return False

    print("Enter q to quit at anytime...")

        g = input("Enter a number: ")
        if g == 'q':

            print(f"'{g}' is an integer")
