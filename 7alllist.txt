while True:
    s = input("Enter several numbers or q to quit:\n")
    if s == "q":
        break
    s = s.split(' ')  # list of strings
    nums = []    # empty list
    for i in range(len(s)):
        try:   # try to convert the string to a float(execute the code)
            v = float(s[i])
            nums.append(v)   # add an item to the end of the list
        except ValueError:   #skip item if it is not a valid number
            v
    print(nums)

    if s == "q":
        break
    s = s.split(' ')
    nums = []
    for i in range(len(s)):
        try:
            v = float(s[i])
            nums.append(v)
        except ValueError:
            v
    print(nums)
    max = min = nums[0]
    for value in nums:
        if min > value:
            min = value
        if max < value:
            max = value
    print(f"The min value is {min}. ")
    print(f"The max value is {max}. ")

import re   # import regular expression module
while True:
    s = input("Enter a sentence or q to quit:\n")
    if s == "q":
        break
    words = re.split(r'\W+', s)  # split words by non-word characters
    words.pop(len(words)-1)
    print(f"The list of words is:\n{words}. ")
    words.sort(reverse=True,key=str.lower)  # sort words in descending order(case-insensitive)
    print(f"The reversed sorted list of words is:\n{words}. ")
