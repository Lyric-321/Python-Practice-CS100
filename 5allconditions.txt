-------ex1.py-------
""" In this program
we will read a grade.
If the grade is at least 60,
we will print a message "This is a passing grade."
Otherwise we will print "This is a failing grade."
"""

g = input("Enter a grade between 0 and 100: ")
g = float(g)
if g >= 60:
    print(f"{g} is a passing grade.")
else:
    print(f"{g} is a failing grade.")

""" In this program
we will read a grade.
we will assign a letter grade.
"""

g = input("Enter a grade between 0 and 100: ")

if g >= 90:
    letter = "A"
elif 80 <= g and g < 90:
    letter = "B"
elif 70 <= g and g < 80:
    letter = "C"
elif 60 <= g and g < 70:
    letter = "D"
else:
    letter = "F"

if g >= 60:
    print(f"{g} is a letter grade of {letter}")
    print(f"{g} is a passing grade.")
else:
    print(f"{g} is a letter grade of {letter}")
    print(f"{g} is a failing grade.")

""" In this program
we will print whether the number is positive negative
or neither
"""

print("Enter q to quit: ")
while True:
    x = input("Enter a number x: ")
    if x == "q":
        break
    x = float(x)
    if x > 0:
        print(f"{x} is positive.")
    elif x <0:
        print(f"{x} is negative.")
    else:
        print(f"{x} is neither positive nor negative.")

swap 2 values.
"""

        break
    n1 = float(input("Enter a number n1: "))
    n2 = float(input("Enter a number n2: "))

    print(f"n1 = {n1}, n2 = {n2}")
    t = n1
    n1 = n2
    n2 = t
    print(f"n1 = {n1}, n2 = {n2}")
