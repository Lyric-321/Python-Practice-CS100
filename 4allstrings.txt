-------ex4.py-------
# the bracket operator [] for range
s = input("Enter a string: " )
print( f"You entered the following '{s}'" )

# the chunk of first 5 chars
s1 = s[0:5]

# the chunk of 3 chars starting at pos 5
s2 = s[5:3]

# the chunk of last 4 chars
s3 = s[-4:]

print( f"The substring s[0:5] '{s1}'")
print( f"The substring s[5:3] '{s2}'")
print( f"The substring s[-4:] '{s3}'")


'''
Sample Output
cs100fa2026/4strings/ $ python ex4.py
Enter a string: Today is a nice day.
You entered the following 'Today is a nice day.'
The substring s[0:5] 'Today'
The substring s[5:3] ''
The substring s[-4:] 'day.'
'''
